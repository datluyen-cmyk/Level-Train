    sealed class State { public Dictionary<int,float> angles=new Dictionary<int,float>(); public HashSet<int> removed=new HashSet<int>(); public HashSet<int> keys=new HashSet<int>(); public int moves; public State Copy(){var s=new State{moves=moves,angles=new Dictionary<int,float>(angles),removed=new HashSet<int>(removed),keys=new HashSet<int>(keys)};return s;} }
    sealed class Drag { public int id; public bool slider; public bool beganLinked; public Vector2 startWorld,lastWorld; public float startAngle,lastPointerAngle,delta; public bool valid; }
    static readonly Color Cream=new Color(.98f,.95f,.88f), Ink=new Color(.30f,.20f,.14f), Brown=new Color(.56f,.34f,.21f);

    Canvas canvas; RectTransform uiRoot; ScreenMode mode; Stack<State> undo=new Stack<State>(); Drag drag; int levelIndex; int maxUnlocked; bool soundOn=true;

    void Awake(){Application.targetFrameRate=60;maxUnlocked=Mathf.Max(0,PlayerPrefs.GetInt("rr_max_level",0));CreateCamera();CreateWorld();CreateCanvas();LoadPack();if(pack!=null&&pack.levels.Count>0)ShowHome();else ShowError("LEVEL PACK MISSING","Place generated_levels.json in Resources/Levels.");LoadAudio();}
    void CreateCamera(){gameCamera=Camera.main;if(gameCamera==null){var go=new GameObject("Main Camera");go.tag="MainCamera";gameCamera=go.AddComponent<Camera>();}gameCamera.orthographic=true;gameCamera.orthographicSize=8;gameCamera.transform.position=new Vector3(0,0,-20);gameCamera.backgroundColor=Cream;gameCamera.clearFlags=CameraClearFlags.SolidColor;gameCamera.allowHDR=false;}
    void CreateWorld(){var go=new GameObject("Puzzle Board");world=go.AddComponent<RotateRingsWorldView>();}
    void CreateCanvas(){var go=new GameObject("Rotate Rings UI");canvas=go.AddComponent<Canvas>();canvas.renderMode=RenderMode.ScreenSpaceOverlay;canvas.sortingOrder=20;var scaler=go.AddComponent<CanvasScaler>();scaler.uiScaleMode=CanvasScaler.ScaleMode.ScaleWithScreenSize;scaler.referenceResolution=new Vector2(1080,1920);scaler.matchWidthOrHeight=.45f;go.AddComponent<GraphicRaycaster>();uiRoot=go.GetComponent<RectTransform>();if(EventSystem.current==null){var e=new GameObject("EventSystem");e.AddComponent<EventSystem>();
#if ENABLE_INPUT_SYSTEM
            e.AddComponent<UnityEngine.InputSystem.UI.InputSystemUIInputModule>();

            e.AddComponent<StandaloneInputModule>();

        }}
    void LoadPack(){var file=Resources.Load<TextAsset>("Levels/generated_levels");if(file==null)return;try{pack=JsonUtility.FromJson<RotateRingsLevelPack>(file.text);if(pack==null||pack.schemaVersion!=1||pack.levels==null)pack=null;else pack.levels.Sort((a,b)=>a.id.CompareTo(b.id));}catch(Exception e){Debug.LogError("Could not load level pack: "+e);pack=null;}}
    void LoadAudio(){music=gameObject.AddComponent<AudioSource>();music.loop=true;music.volume=.24f;var clip=Resources.Load<AudioClip>("Audio/rr_bgm_level2");if(clip!=null){music.clip=clip;music.Play();}sfx=gameObject.AddComponent<AudioSource>();sfx.volume=.52f;}
    void Update(){if(mode==ScreenMode.Game)PollInput();}
    void ClearUI(){if(panel!=null)Destroy(panel);panel=null;toastText=null;titleText=null;movesText=null;}
    GameObject MakePanel(string name,Color color,RectTransform parent=null){var g=new GameObject(name,typeof(RectTransform),typeof(Image));g.transform.SetParent(parent??uiRoot,false);var i=g.GetComponent<Image>();i.color=color;i.raycastTarget=false;return g;}
    RectTransform Rect(GameObject go)=>go.GetComponent<RectTransform>();
    void Full(GameObject go){var r=Rect(go);r.anchorMin=Vector2.zero;r.anchorMax=Vector2.one;r.offsetMin=Vector2.zero;r.offsetMax=Vector2.zero;}
    Text Label(string value,Transform parent,Vector2 anchor,Vector2 size,int fontSize,Color color,TextAnchor align=TextAnchor.MiddleCenter,FontStyle style=FontStyle.Bold){var go=new GameObject("Text",typeof(RectTransform),typeof(Text));go.transform.SetParent(parent,false);var t=go.GetComponent<Text>();t.font=Resources.GetBuiltinResource<Font>("LegacyRuntime.ttf");t.text=value;t.fontSize=fontSize;t.color=color;t.alignment=align;t.fontStyle=style;t.horizontalOverflow=HorizontalWrapMode.Wrap;t.verticalOverflow=VerticalWrapMode.Truncate;t.raycastTarget=false;var r=Rect(go);r.anchorMin=r.anchorMax=anchor;r.sizeDelta=size;return t;}
    Button Btn(string text,Transform parent,Vector2 anchor,Vector2 size,Action click,Color? color=null,Sprite sprite=null){var go=new GameObject("Button - "+text,typeof(RectTransform),typeof(Image),typeof(Button));go.transform.SetParent(parent,false);var im=go.GetComponent<Image>();im.color=color??new Color(.99f,.76f,.20f);im.sprite=sprite??Art("home_entry_bg02");im.type=Image.Type.Simple;var b=go.GetComponent<Button>();b.targetGraphic=im;b.onClick.AddListener(()=>click?.Invoke());var r=Rect(go);r.anchorMin=r.anchorMax=anchor;r.sizeDelta=size;var label=Label(text,go.transform,new Vector2(.5f,.5f),size*.88f,Mathf.Clamp((int)(size.y*.32f),22,48),Ink);label.raycastTarget=false;return b;}
    Sprite Art(string prefix){if(artCache.TryGetValue(prefix,out var cached))return cached;string alias=prefix;switch(prefix){case "btn_level_yellow":alias="level_button";break;case "home_entry_bg02":alias="home_entry_button";break;case "home_entry_bg01":alias="home_entry_round_button";break;}var all=Resources.LoadAll<Texture2D>("Art");foreach(var t in all)if(t.name.IndexOf(alias,StringComparison.OrdinalIgnoreCase)>=0){var s=Sprite.Create(t,new Rect(0,0,t.width,t.height),new Vector2(.5f,.5f),100);artCache[prefix]=s;return s;}return null;}
    void SetSafeBoard(){if(level==null)return;float minX=float.MaxValue,maxX=float.MinValue,minY=float.MaxValue,maxY=float.MinValue;foreach(var p in level.parts){float r=RotateRingsRules.IsSlider(p)?p.length*.5f:p.radius+p.width;minX=Mathf.Min(minX,p.x-r);maxX=Mathf.Max(maxX,p.x+r);minY=Mathf.Min(minY,p.y-r);maxY=Mathf.Max(maxY,p.y+r);}float h=Mathf.Max(2,maxY-minY),w=Mathf.Max(2,maxX-minX);float aspect=(float)Screen.width/Mathf.Max(1,Screen.height);gameCamera.orthographicSize=Mathf.Max(h*.62f,w/(2*Mathf.Max(aspect,.2f)))*1.23f;gameCamera.transform.position=new Vector3((minX+maxX)*.5f,(minY+maxY)*.5f,-20);}

    void ShowHome(){mode=ScreenMode.Home;world.gameObject.SetActive(false);ClearUI();panel=MakePanel("Home Screen",Cream);Full(panel);Label("ROTATE RINGS",panel.transform,new Vector2(.5f,.83f),new Vector2(900,150),72,Ink);Label("Untangle the colorful rings",panel.transform,new Vector2(.5f,.76f),new Vector2(850,80),33,Brown,TextAnchor.MiddleCenter,FontStyle.Normal);
        Btn("PLAY",panel.transform,new Vector2(.5f,.59f),new Vector2(460,145),()=>StartLevel(pack.levels[Mathf.Clamp(maxUnlocked,0,pack.levels.Count-1)].id),new Color(.99f,.78f,.26f),Art("home_entry_bg02"));
        Btn("CHOOSE LEVEL",panel.transform,new Vector2(.5f,.47f),new Vector2(470,118),ShowLevels,new Color(1f,.89f,.69f),Art("home_entry_bg02"));
        Label("300 generated puzzles  •  progress saved",panel.transform,new Vector2(.5f,.20f),new Vector2(880,70),25,new Color(.53f,.43f,.34f),TextAnchor.MiddleCenter,FontStyle.Normal);gameCamera.backgroundColor=Cream;}
    void ShowLevels(){mode=ScreenMode.Levels;world.gameObject.SetActive(false);ClearUI();panel=MakePanel("Level Select",Cream);Full(panel);Label("CHOOSE A LEVEL",panel.transform,new Vector2(.5f,.91f),new Vector2(820,100),52,Ink);Btn("‹  HOME",panel.transform,new Vector2(.15f,.91f),new Vector2(180,82),ShowHome,new Color(1f,.89f,.69f));
        var scrollGo=new GameObject("Level Grid",typeof(RectTransform),typeof(ScrollRect),typeof(Image),typeof(RectMask2D));scrollGo.transform.SetParent(panel.transform,false);var sr=scrollGo.GetComponent<ScrollRect>();var ir=scrollGo.GetComponent<Image>();ir.color=new Color(1,1,1,.01f);var srt=Rect(scrollGo);srt.anchorMin=new Vector2(.07f,.08f);srt.anchorMax=new Vector2(.93f,.83f);srt.offsetMin=srt.offsetMax=Vector2.zero;
        var content=new GameObject("Content",typeof(RectTransform),typeof(GridLayoutGroup),typeof(ContentSizeFitter));content.transform.SetParent(scrollGo.transform,false);var cr=Rect(content);cr.anchorMin=new Vector2(0,1);cr.anchorMax=new Vector2(1,1);cr.pivot=new Vector2(.5f,1);cr.anchoredPosition=Vector2.zero;var grid=content.GetComponent<GridLayoutGroup>();grid.cellSize=new Vector2(145,125);grid.spacing=new Vector2(20,22);grid.padding=new RectOffset(8,8,10,16);grid.constraint=GridLayoutGroup.Constraint.FixedColumnCount;grid.constraintCount=5;content.GetComponent<ContentSizeFitter>().verticalFit=ContentSizeFitter.FitMode.PreferredSize;sr.content=cr;sr.horizontal=false;sr.vertical=true;
        for(int i=0;i<pack.levels.Count;i++){int ix=i;int no=i+1;var b=Btn(""+no,content.transform,new Vector2(.5f,.5f),grid.cellSize,()=>StartLevel(pack.levels[ix].id),new Color(.99f,.88f,.54f),Art("btn_level_yellow"));var br=Rect(b.gameObject);br.anchorMin=br.anchorMax=new Vector2(.5f,.5f);br.sizeDelta=grid.cellSize;}
        gameCamera.backgroundColor=Cream;}
    void StartLevel(int requested){int index=pack.levels.FindIndex(l=>l.id==requested);if(index<0)index=Mathf.Clamp(requested,0,pack.levels.Count-1);levelIndex=index;level=pack.levels[levelIndex];state=new State();undo.Clear();foreach(var p in level.parts)state.angles[p.id]=p.angle;world.gameObject.SetActive(true);world.Build(level);world.SetHint(-1);SetSafeBoard();mode=ScreenMode.Game;RenderGameUI();UpdateWorld();}
    void RenderGameUI(){ClearUI();panel=MakePanel("Gameplay HUD",Color.clear);Full(panel);titleText=Label("LEVEL "+(levelIndex+1),panel.transform,new Vector2(.5f,.955f),new Vector2(600,74),38,Ink);movesText=Label("MOVES  0",panel.transform,new Vector2(.5f,.905f),new Vector2(500,60),26,Brown,TextAnchor.MiddleCenter,FontStyle.Normal);Label("Open a ring gap to free its pin  •  slide bars along the arrow",panel.transform,new Vector2(.5f,.855f),new Vector2(950,58),21,Brown,TextAnchor.MiddleCenter,FontStyle.Normal);
        Btn("HOME",panel.transform,new Vector2(.14f,.95f),new Vector2(165,72),ShowHome,new Color(1f,.88f,.65f));Btn("↻",panel.transform,new Vector2(.86f,.95f),new Vector2(90,72),Restart,new Color(1f,.88f,.65f));
        Btn("UNDO",panel.transform,new Vector2(.25f,.075f),new Vector2(210,90),Undo,new Color(.96f,.84f,.65f));Btn("HINT",panel.transform,new Vector2(.5f,.075f),new Vector2(210,90),Hint,new Color(.99f,.73f,.20f));Btn("MUSIC",panel.transform,new Vector2(.75f,.075f),new Vector2(210,90),ToggleSound,new Color(.96f,.84f,.65f));}
    void UpdateWorld(){if(world==null||state==null)return;var active=new List<bool>();foreach(var l in level.links){var s=FindPart(l.source);var t=FindPart(l.target);active.Add(RotateRingsRules.IsLinkActive(l,s,t,state.angles[l.source],state.angles[l.target],state.removed.Contains(l.source),state.removed.Contains(l.target)));}world.ApplyState(state.angles,state.removed,state.keys,active);if(movesText!=null)movesText.text="MOVES  "+state.moves+(level.parMoves>0?"   •   PAR "+level.parMoves:"");}
    RotateRingsPart FindPart(int id)=>level.parts.Find(p=>p.id==id);

    void PollInput(){if(drag==null){if(Down(out int pid,out Vector2 screen)){if(EventSystem.current!=null&&EventSystem.current.IsPointerOverGameObject(pid))return;Vector2 w=gameCamera.ScreenToWorldPoint(screen);if(!world.TryHit(w,out int id))return;var p=FindPart(id);if(p==null||state.removed.Contains(id))return;bool slider=RotateRingsRules.IsSlider(p);if(p.lockKey!=0&&!state.keys.Contains(p.lockKey)){Wobble(id);return;}if(!slider&&HasActiveOutgoing(id)){Wobble(id);return;}world.SetHint(-1);drag=new Drag{id=id,slider=slider,beganLinked=!slider&&HasActiveIncident(id),startWorld=w,lastWorld=w,startAngle=state.angles[id],lastPointerAngle=Mathf.Atan2(w.y-p.y,w.x-p.x)*Mathf.Rad2Deg,valid=true};PushUndo();}return;}
        if(Held(out Vector2 current)){Vector2 w=gameCamera.ScreenToWorldPoint(current);var p=FindPart(drag.id);if(p==null){drag=null;return;}if(drag.slider){Vector2 axis=new Vector2(Mathf.Cos(p.axis*Mathf.Deg2Rad),Mathf.Sin(p.axis*Mathf.Deg2Rad));drag.delta=Vector2.Dot(w-drag.startWorld,axis);world.SetSliderOffset(drag.id,drag.delta);UpdateWorld();}
            else {float a=Mathf.Atan2(w.y-p.y,w.x-p.x)*Mathf.Rad2Deg;float d=Mathf.DeltaAngle(drag.lastPointerAngle,a);drag.lastPointerAngle=a;float before=state.angles[p.id];state.angles[p.id]=before+d;if(HasActiveOutgoing(p.id)){state.angles[p.id]=before;Wobble(p.id);}else{drag.delta+=d;UpdateWorld();}}drag.lastWorld=w;return;}
        if(Up()){FinishDrag();}}
    bool Down(out int id,out Vector2 p){id=-1;p=default;
#if ENABLE_INPUT_SYSTEM
        if(Mouse.current!=null&&Mouse.current.leftButton.wasPressedThisFrame){p=Mouse.current.position.ReadValue();return true;}if(Touchscreen.current!=null){foreach(var t in Touchscreen.current.touches)if(t.press.wasPressedThisFrame){id=t.touchId.ReadValue();p=t.position.ReadValue();return true;}}
#else
        if(Input.GetMouseButtonDown(0)){p=Input.mousePosition;return true;}if(Input.touchCount>0&&Input.GetTouch(0).phase==TouchPhase.Began){id=Input.GetTouch(0).fingerId;p=Input.GetTouch(0).position;return true;}
#endif
        return false;}
    bool Held(out Vector2 p){p=default;
#if ENABLE_INPUT_SYSTEM
        if(Mouse.current!=null&&Mouse.current.leftButton.isPressed){p=Mouse.current.position.ReadValue();return true;}if(Touchscreen.current!=null){foreach(var t in Touchscreen.current.touches)if(t.press.isPressed){p=t.position.ReadValue();return true;}}
#else
        if(Input.GetMouseButton(0)){p=Input.mousePosition;return true;}if(Input.touchCount>0){p=Input.GetTouch(0).position;return true;}
#endif
        return false;}
    bool Up(){
#if ENABLE_INPUT_SYSTEM
        if(Mouse.current!=null&&Mouse.current.leftButton.wasReleasedThisFrame)return true;if(Touchscreen.current!=null){foreach(var t in Touchscreen.current.touches)if(t.press.wasReleasedThisFrame)return true;}
#else
        if(Input.GetMouseButtonUp(0)||Input.touchCount==0)return true;
#endif
        return false;}
    void FinishDrag(){if(drag==null)return;var d=drag;drag=null;if(d.slider){var p=FindPart(d.id);if(p!=null&&d.delta>=p.radius*1.25f&&!HasActiveOutgoing(d.id)){state.moves++;RemovePart(d.id);Cascade(d.id,false);UpdateWorld();CheckWin();}else {world.SetSliderOffset(d.id,0);UpdateWorld();if(undo.Count>0)undo.Pop();if(p!=null)Wobble(d.id);}return;}
        if(Mathf.Abs(d.delta)<.1f){state=undo.Pop();UpdateWorld();return;}state.moves++;bool released=TryRelease(d.id,d.delta,d.beganLinked);var touched=FindPart(d.id);Cascade(d.id,released||touched==null||!touched.collectible);UpdateWorld();CheckWin();}
    bool HasActiveOutgoing(int id){for(int i=0;i<level.links.Count;i++){var l=level.links[i];if(l.source==id&&IsActive(l))return true;}return false;}
    bool HasActiveIncident(int id){for(int i=0;i<level.links.Count;i++){var l=level.links[i];if((l.source==id||l.target==id)&&IsActive(l))return true;}return false;}
    bool IsActive(RotateRingsLink l)=>RotateRingsRules.IsLinkActive(l,FindPart(l.source),FindPart(l.target),state.angles[l.source],state.angles[l.target],state.removed.Contains(l.source),state.removed.Contains(l.target));
    bool IsLocked(RotateRingsPart p)=>p.lockKey!=0&&!state.keys.Contains(p.lockKey);
    bool TryRelease(int id,float gestureDelta,bool beganLinked){var p=FindPart(id);if(p==null||!p.collectible)return false;float threshold=beganLinked?2f:15f;if(!IsLocked(p)&&!HasActiveIncident(id)&&RotateRingsRules.MeetsGestureThreshold(gestureDelta,threshold)){RemovePart(id);return true;}else if(HasActiveIncident(id)||IsLocked(p))Wobble(id);return false;}
    void RemovePart(int id){var p=FindPart(id);if(p==null)return;world.BeginRelease(id);state.removed.Add(id);if(p.key!=0)state.keys.Add(p.key);PlayBreak(p.color);}
    void Cascade(int touched,bool includeTouched=true){bool changed=true;int guard=0;while(changed&&guard++<level.parts.Count){changed=false;foreach(var p in level.parts){if(state.removed.Contains(p.id)||!p.collectible||RotateRingsRules.IsSlider(p)||IsLocked(p)||HasActiveIncident(p.id))continue;bool related=includeTouched&&p.id==touched;foreach(var l in level.links)if((l.source==touched&&l.target==p.id)||(l.target==touched&&l.source==p.id))related=true;if(related){RemovePart(p.id);changed=true;}}}}
    void PushUndo(){if(state!=null)undo.Push(state.Copy());}
    void Undo(){if(undo.Count==0){Toast("NOTHING TO UNDO");return;}state=undo.Pop();foreach(var p in level.parts)if(RotateRingsRules.IsSlider(p))world.SetSliderOffset(p.id,0);UpdateWorld();}
    void Restart(){StartLevel(level.id);}
    void Hint(){int hintPart=-1;string instruction=null;
        if(level.solution!=null)foreach(var g in level.solution){if(state.removed.Contains(g.part))continue;var p=FindPart(g.part);if(p==null)continue;if(string.Equals(g.kind,"slide",StringComparison.OrdinalIgnoreCase)){if(RotateRingsRules.IsSlider(p)&&!IsLocked(p)&&!HasActiveOutgoing(p.id)){hintPart=p.id;instruction="Slide bar "+p.id+" out along its guide";break;}}
            else if(string.Equals(g.kind,"rotate",StringComparison.OrdinalIgnoreCase)&&!RotateRingsRules.IsSlider(p)&&!IsLocked(p)&&!HasActiveOutgoing(p.id)){float delta=Mathf.DeltaAngle(state.angles[p.id],g.angle);float threshold=HasActiveIncident(p.id)?2f:15f;if(RotateRingsRules.MeetsGestureThreshold(delta,threshold)&&(!p.collectible||CanReleaseAt(p,g.angle))){hintPart=p.id;instruction="Rotate ring "+p.id+" to "+Mathf.RoundToInt(Mathf.Repeat(g.angle,360))+"°";break;}}}
        if(hintPart<0){foreach(var p in level.parts){if(state.removed.Contains(p.id)||IsLocked(p))continue;if(RotateRingsRules.IsSlider(p)){if(!HasActiveOutgoing(p.id)){hintPart=p.id;instruction="Slide bar "+p.id+" out along its guide";break;}continue;}if(HasActiveOutgoing(p.id))continue;
                for(int step=1;step<=24;step++){float target=state.angles[p.id]+step*15f;bool useful=p.collectible?CanReleaseAt(p,target):ChangesLinkActivityAt(p,target);if(useful){hintPart=p.id;instruction="Rotate ring "+p.id+" toward "+Mathf.RoundToInt(Mathf.Repeat(target,360))+"°";break;}}if(hintPart>=0)break;}}
        if(hintPart<0){Toast("TRY OPENING A CONNECTED RING");return;}world.SetHint(hintPart);Toast(instruction);}
    bool CanReleaseAt(RotateRingsPart p,float target){float old=state.angles[p.id];state.angles[p.id]=target;bool free=!HasActiveIncident(p.id);state.angles[p.id]=old;return free;}
    bool ChangesLinkActivityAt(RotateRingsPart p,float target){var before=new bool[level.links.Count];for(int i=0;i<before.Length;i++)before[i]=IsActive(level.links[i]);float old=state.angles[p.id];state.angles[p.id]=target;bool changed=false;for(int i=0;i<before.Length;i++)if(before[i]!=IsActive(level.links[i])){changed=true;break;}state.angles[p.id]=old;return changed;}
    void ToggleSound(){soundOn=!soundOn;if(music!=null)music.mute=!soundOn;if(sfx!=null)sfx.mute=!soundOn;Toast(soundOn?"SOUND ON":"SOUND OFF");}
    void Wobble(int id){if(world!=null&&state!=null){var active=new List<bool>();foreach(var l in level.links)active.Add(IsActive(l));world.ApplyState(state.angles,state.removed,state.keys,active,true,id);}Toast("THIS RING IS STILL CONNECTED");}
    void PlayBreak(int color){if(!soundOn||sfx==null)return;string n=color%3==0?"rr_sfx_wood_break_1":color%3==1?"rr_sfx_rock_break_1":"rr_sfx_jelly_break_1";var clip=Resources.Load<AudioClip>("Audio/"+n);if(clip!=null)sfx.PlayOneShot(clip);}
    void CheckWin(){foreach(var p in level.parts)if(p.collectible&&!state.removed.Contains(p.id))return;PlayerPrefs.SetInt("rr_max_level",Mathf.Max(maxUnlocked,Mathf.Min(levelIndex+1,pack.levels.Count-1)));maxUnlocked=Mathf.Max(maxUnlocked,Mathf.Min(levelIndex+1,pack.levels.Count-1));PlayerPrefs.Save();ShowWin();}
    void ShowWin(){mode=ScreenMode.Win;world.gameObject.SetActive(false);ClearUI();panel=MakePanel("Level Clear",new Color(.99f,.95f,.87f));Full(panel);Label("PUZZLE COMPLETE!",panel.transform,new Vector2(.5f,.67f),new Vector2(900,160),62,Ink);Label("Cleared in "+state.moves+" moves",panel.transform,new Vector2(.5f,.57f),new Vector2(800,90),34,Brown,TextAnchor.MiddleCenter,FontStyle.Normal);Btn("NEXT LEVEL  ›",panel.transform,new Vector2(.5f,.43f),new Vector2(490,130),()=>{if(levelIndex+1<pack.levels.Count)StartLevel(pack.levels[levelIndex+1].id);else ShowHome();},new Color(.99f,.75f,.16f));Btn("LEVELS",panel.transform,new Vector2(.5f,.32f),new Vector2(390,100),ShowLevels,new Color(1,.88f,.67f));}
    void ShowError(string header,string body){mode=ScreenMode.Home;ClearUI();panel=MakePanel("Setup Required",Cream);Full(panel);Label(header,panel.transform,new Vector2(.5f,.57f),new Vector2(900,140),52,Ink);Label(body,panel.transform,new Vector2(.5f,.47f),new Vector2(900,150),30,Brown,TextAnchor.MiddleCenter,FontStyle.Normal);}
    void Toast(string text){if(toastText==null)toastText=Label(text,uiRoot,new Vector2(.5f,.17f),new Vector2(820,72),25,Ink);else toastText.text=text;toastText.gameObject.SetActive(true);CancelInvoke(nameof(HideToast));Invoke(nameof(HideToast),1.25f);}void HideToast(){if(toastText!=null)toastText.gameObject.SetActive(false);}
}
